一、项目概述
1.1 产品定位
元生 AI 生态健康饮食积分系统是一个集健康评估、AI 饮食建议、积分激励、线下服务于一体的一站式健康管理平台。通过 AI 技术分析用户健康状况，生成个性化饮食改善方案，结合积分激励机制和线下服务网络，实现用户健康管理和商业转化。
核心特色：
🎯 AI 驯化：5 层控制体系，防止编造数据、医疗越界
🪝 危机钩子：2 份报告设计，引导到店转化
👴 适老化：大字体、大按钮、简化操作
🏅 荣誉体系：等级勋章、惊喜激励
🛡️ 隐私保护：支持非实名注册

// 检查本月评估次数
async function checkAssessmentLimit(userId) {
  const currentDate = new Date();
  const currentMonth = currentDate.getMonth();
  const currentYear = currentDate.getFullYear();
  
  const assessmentRecord = await db.assessments.findOne({
    user_id: userId,
    year: currentYear,
    month: currentMonth
  });
  
  if (assessmentRecord && assessmentRecord.count >= 1) {
    return {
      allowed: false,
      message: '您本月的评估次数已用完，请于下月再来',
      nextAvailableDate: new Date(currentYear, currentMonth + 1, 1)
    };
  }
  
  return { allowed: true };
}
3.5 流失预警机制
// 检测 3 天无活跃用户
async function checkInactiveUsers() {
  const threeDaysAgo = new Date();
  threeDaysAgo.setDate(threeDaysAgo.getDate() - 3);
  
  // 查询 3 天无打卡、无课程学习的会员
  const inactiveUsers = await db.users.find({
    is_member: true,
    last_active_date: { $lt: threeDaysAgo }
  });
  
  for (const user of inactiveUsers) {
    // 通知代理商
    await notifyAgent(user.agent_id, {
      type: 'user_inactive',
      user_id: user.id,
      user_name: user.name,
      days_inactive: 3,
      message: `您的用户${user.name}已 3 天未活跃，请及时跟进`
    });
    
    // 通知服务商
    if (user.service_provider_id) {
      await notifyServiceProvider(user.service_provider_id, {
        type: 'user_inactive',
        user_id: user.id,
        user_name: user.name,
        days_inactive: 3,
        message: `您的客户${user.name}已 3 天未活跃，请及时跟进`
      });
    }
  }
}

// 每日凌晨执行
cron.schedule('0 2 * * *', checkInactiveUsers);

# 角色设定

你是一位拥有 15 年临床经验的注册营养师，擅长根据用户的饮食生活习惯分析潜在健康风险。

# 工作原则

1. **真实性原则**：只基于用户提供的真实信息分析，不编造数据
2. **个性化原则**：每个建议必须与用户具体情况直接关联
3. **谨慎性原则**：使用"可能"、"风险增加"等谨慎表述
4. **合规原则**：不做疾病诊断，不承诺治疗效果

# 禁止行为

❌ 编造用户未提供的体检数值
❌ 做出疾病诊断（如"您患有糖尿病"）
❌ 承诺治疗效果（如"7 天治愈"）
❌ 使用恐吓性语言（如"会得癌症"）
❌ 使用绝对化表述（如"一定会"、"肯定"）

# 必须包含

✅ 基于用户真实数据的分析
✅ 谨慎的风险表述（"可能"、"风险增加"）
✅ 免责声明

5.5 第 4 层：自动验证
敏感词检测规则
// 禁止出现的表述（医疗红线）
const FORBIDDEN_PATTERNS = [
  { pattern: /治疗.*疾病/, message: '禁止使用"治疗疾病"表述' },
  { pattern: /治愈/, message: '禁止使用"治愈"表述' },
  { pattern: /诊断.*为/, message: '禁止使用"诊断为"表述' },
  { pattern: /处方/, message: '禁止使用"处方"表述' },
  { pattern: /代替.*药物/, message: '禁止使用"代替药物"表述' },
  { pattern: /疗效/, message: '禁止使用"疗效"表述' },
  { pattern: /根治/, message: '禁止使用"根治"表述' },
  { pattern: /停药/, message: '禁止使用"停药"表述' },
  { pattern: /保证.*天好转/, message: '禁止承诺效果' },
  { pattern: /肯定.*改善/, message: '禁止绝对化表述' }
];

// 验证函数
function checkSensitiveWords(output) {
  const errors = [];
  
  FORBIDDEN_PATTERNS.forEach(({ pattern, message }) => {
    if (pattern.test(output)) {
      errors.push(`[医疗红线] ${message}`);
    }
  });
  
  return errors;
}

7.3 饮食图片验证
// 调用阿里云内容安全 API 验证图片
async function validateFoodImage(imageUrl) {
  const result = await aliyunContentModeration.imageDetection({
    imageUrl: imageUrl,
    scenes: ['porn', 'terrorism', 'ad', 'live', 'logo', 'qrcode']
  });
  
  // 检测是否包含违规内容
  if (result.label === 'porn' || result.label === 'terrorism') {
    return {
      valid: false,
      message: '图片包含违规内容，请上传饮食相关图片'
    };
  }
  
  // 检测是否包含食物（使用 AI 图像识别）
  const foodDetection = await aliyunImageRecognition.detect({
    imageUrl: imageUrl,
    categories: ['food', 'dish', 'fruit', 'vegetable']
  });
  
  if (foodDetection.confidence < 0.6) {
    return {
      valid: false,
      message: '请上传饮食相关图片'
    };
  }
  
  return { valid: true };
}
